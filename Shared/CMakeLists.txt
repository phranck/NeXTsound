cmake_minimum_required(VERSION 3.22)
project(NeXTsoundProtocol VERSION 0.1.0 LANGUAGES C)

add_library(nxs_protocol STATIC Sources/NeXTsoundProtocol/protocol.c)
target_include_directories(nxs_protocol PUBLIC Sources/NeXTsoundProtocol/include)
target_compile_features(nxs_protocol PUBLIC c_std_11)
target_compile_options(nxs_protocol PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_library(nxs_dsp STATIC Sources/NeXTsoundDSP/dsp.c)
target_include_directories(nxs_dsp PUBLIC Sources/NeXTsoundDSP/include)
target_link_libraries(nxs_dsp PUBLIC nxs_protocol m)
target_compile_features(nxs_dsp PUBLIC c_std_11)
target_compile_options(nxs_dsp PRIVATE -Wall -Wextra -Wpedantic -Werror)

if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
    option(NXS_ENABLE_SANITIZERS "Enable host address and undefined-behavior sanitizers" ON)
    if(NXS_ENABLE_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(nxs_protocol PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(nxs_protocol PUBLIC -fsanitize=address,undefined)
    endif()
    if(BUILD_TESTING)
        function(nxs_add_host_test name source library)
            set(test_target "nxs_${name}_tests")
            add_executable("${test_target}" "${source}")
            target_link_libraries("${test_target}" PRIVATE "${library}")
            target_include_directories("${test_target}" PRIVATE "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/Tests/Support")
            target_compile_options("${test_target}" PRIVATE -Wall -Wextra -Wpedantic -Werror)
            add_test(NAME "${name}" COMMAND "${test_target}" ${ARGN})
            set_tests_properties("${name}" PROPERTIES TIMEOUT 30)
        endfunction()
        nxs_add_host_test(protocol Tests/CProtocolTests/protocol_tests.c nxs_protocol
            "${CMAKE_CURRENT_SOURCE_DIR}/Tests/NeXTsoundCoreTests/Fixtures/capabilities-request.hex")
        nxs_add_host_test(dsp Tests/CDSPTests/dsp_tests.c nxs_dsp)
        add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../Firmware/Core" "firmware-core")
        nxs_add_host_test(device "${CMAKE_CURRENT_SOURCE_DIR}/../Firmware/Tests/device_tests.c" nxs_device)
        find_package(ZLIB REQUIRED)
        nxs_add_host_test(storage "${CMAKE_CURRENT_SOURCE_DIR}/../Firmware/Tests/storage_tests.c" nxs_storage
            "${CMAKE_CURRENT_SOURCE_DIR}/../Firmware/Tests/Fixtures/storage-record-v1.hex")
        target_link_libraries(nxs_storage_tests PRIVATE ZLIB::ZLIB)
        nxs_add_host_test(flash_geometry "${CMAKE_CURRENT_SOURCE_DIR}/../Firmware/Tests/flash_geometry_tests.c" nxs_storage)
        find_package(Threads REQUIRED)
        nxs_add_host_test(audio "${CMAKE_CURRENT_SOURCE_DIR}/../Firmware/Tests/audio_tests.c" nxs_device)
        target_link_libraries(nxs_audio_tests PRIVATE Threads::Threads)
        nxs_add_host_test(render "${CMAKE_CURRENT_SOURCE_DIR}/../Firmware/Tests/render_tests.c" nxs_device)
        option(NXS_TEST_FIRMWARE "Test built firmware descriptors and PIO code" OFF)
        if(NXS_TEST_FIRMWARE)
            add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../Firmware/Tests" "firmware-tests")
        endif()
    endif()
endif()
